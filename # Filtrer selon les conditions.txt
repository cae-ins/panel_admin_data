age_unique = panel_solde_df.drop_duplicates(subset=["MATRICULE", "DATE_COLLECTE", "AGE"])

# Étape 2 : Vérifier s'il existe plusieurs âges différents pour un même matricule et une même période
age_incoherent = (
    age_unique.groupby(["MATRICULE", "DATE_COLLECTE"])["AGE"]
    .nunique()
    .reset_index(name="nb_age_distincts")
)

# Étape 3 : Filtrer les cas incohérents
problemes_age = age_incoherent[age_incoherent["nb_age_distincts"] > 1]

# Étape 4 : Afficher un message selon le résultat
if problemes_age.empty:
    print("✅ Aucun problème détecté — les âges sont cohérents par matricule et période.")
else:
    print("⚠️ Matricules ayant plusieurs âges enregistrés pour une même période :")
    print(problemes_age)
